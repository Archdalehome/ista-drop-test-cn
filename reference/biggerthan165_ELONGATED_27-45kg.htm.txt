    <script>
        // 获取所有步骤行
        const rows = document.querySelectorAll('#procedureTable tbody tr');
        let currentStep = 0;

        // 为每个步骤行添加点击事件监听器
        rows.forEach((row, index) => {
            row.addEventListener('click', () => {
                // 移除所有行的高亮状态
                rows.forEach(r => r.classList.remove('active'));
                
                // 为当前点击的行添加高亮状态
                row.classList.add('active');
                
                // 更新当前步骤
                currentStep = index;
            });
        });

        // 键盘导航支持
        document.addEventListener('keydown', (e) => {
            // 移除所有行的高亮状态
            rows.forEach(r => r.classList.remove('active'));
            
            if (e.key === 'ArrowDown' && currentStep < rows.length - 1) {
                // 向下箭头，移动到下一步
                currentStep++;
            } else if (e.key === 'ArrowUp' && currentStep > 0) {
                // 向上箭头，移动到上一步
                currentStep--;
            }
            
            // 为当前步骤添加高亮状态
            rows[currentStep].classList.add('active');
            
            // 确保高亮的步骤在视窗中可见
            rows[currentStep].scrollIntoView({behavior: 'smooth', block: 'center'});
        });
    </script>
